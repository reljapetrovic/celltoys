> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A `candle/` toy: a canvas candle flame that sways with blowing into the mic and goes out when blown hard; tap relights.

**Architecture:** Two pure, tested modules (`blow.js` detector, `flame.js` model) feed a Canvas 2D renderer; `mic.js` wraps getUserMedia + AnalyserNode; `main.js` wires the start screen, loop and settings sheet. Reuses `shared/`.

**Tech Stack:** Vite multi-page (existing), Vitest, Web Audio API, Canvas 2D.

**Spec:** `docs/superpowers/specs/2026-10-07-candle-design.md`

## Global Constraints

- Detector constants: low band 400 Hz, floor rise 0.002/s, dead zone 0.01, full scale 0.2/sensitivity, gate (lowFraction−0.35)/0.3, τ rise 0.05 s, τ fall 0.25 s.
- Flame constants: lean τ 0.08 s, size τ 0.1 s, size = 1 − 0.6·intensity (+0.5·flare), health drains 1/0.35 s above intensity 0.8, refills 1/0.5 s, smoke decays 1/3 s, flare 1/0.4 s, wind flips with p = 0.3·dt while intensity < 0.1.
- `dt` clamped to 0.05 s in the page loop. No white UI. Sensitivity 0.5–3.
- Vite base `/celltoys/`; manifest/icon relative URLs in `public/candle/`.
- Commit after each task with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.

```js
// Blow detector: intensity 0..1 from mic loudness above an adaptive noise
// floor, gated by the share of energy below 400 Hz (blowing is low rumble;
// speech and claps are not). Pure: takes analyser buffers, keeps its own state.

const LOW_HZ = 400, FLOOR_RISE = 0.002, DEAD = 0.01, FULL = 0.2;
const TAU_RISE = 0.05, TAU_FALL = 0.25;
const clamp01 = v => Math.min(1, Math.max(0, v));

export function createBlowDetector({ sampleRate, fftSize, sensitivity = 1 }) {
  const lowBins = Math.max(1, Math.floor(LOW_HZ / (sampleRate / fftSize)));
  let floor = 0, intensity = 0, sens = sensitivity;

  function update(timeData, freqData, dt) {
    let sum = 0;
    for (let i = 0; i < timeData.length; i++) sum += timeData[i] * timeData[i];
    const rms = timeData.length ? Math.sqrt(sum / timeData.length) : 0;

    floor = rms < floor ? rms : Math.min(rms, floor + FLOOR_RISE * dt);
    const loudness = clamp01((rms - floor - DEAD) / (FULL / sens));

    let low = 0, total = 0;
    for (let i = 0; i < freqData.length; i++) {
      const e = freqData[i] * freqData[i];
      total += e;
      if (i < lowBins) low += e;
    }
    const lowFraction = total > 0 ? low / total : 0;
    const gate = clamp01((lowFraction - 0.35) / 0.3);

    const raw = loudness * gate;
    const tau = raw > intensity ? TAU_RISE : TAU_FALL;
    intensity += (raw - intensity) * (1 - Math.exp(-dt / tau));
    return { intensity, loudness, lowFraction, floor };
  }

  return { update, setSensitivity(s) { sens = s; } };
}
```

```js
// Candle flame model. Pure: (state, intensity, dt) -> state.

export const INITIAL_FLAME = Object.freeze({
  lit: true, lean: 0, size: 1, health: 1, smoke: 0, flare: 0, wind: 1,
});

const ease = (cur, target, dt, tau) => cur + (target - cur) * (1 - Math.exp(-dt / tau));
const clamp01 = v => Math.min(1, Math.max(0, v));

export function stepFlame(s, intensity, dt, rand = Math.random) {
  let { lit, lean, size, health, smoke, flare, wind } = s;
  if (dt <= 0) return { ...s };

  if (intensity < 0.1 && rand() < 0.3 * dt) wind = -wind;
  const wobble = (rand() - 0.5) * 0.2 * (0.3 + intensity);
  lean = ease(lean, lit ? wind * intensity + wobble : 0, dt, 0.08);

  flare = Math.max(0, flare - dt / 0.4);
  size = ease(size, lit ? 1 - 0.6 * intensity + 0.5 * flare : 0, dt, 0.1);

  if (lit) health = clamp01(intensity > 0.8 ? health - dt / 0.35 : health + dt / 0.5);
  if (lit && health <= 0) { lit = false; smoke = 1; }
  smoke = Math.max(0, smoke - dt / 3);

  return { lit, lean, size, health, smoke, flare, wind };
}

export function relight(s) {
  return { ...s, lit: true, health: 1, size: 0.2, flare: 1, smoke: 0 };
}
```

```js
// Microphone capture for the blow detector. Disables echo cancellation,
// noise suppression and AGC so blowing is not filtered out.

export async function startMic() {
  if (!navigator.mediaDevices?.getUserMedia) throw new Error('getUserMedia unavailable');
  const stream = await navigator.mediaDevices.getUserMedia({
    audio: { echoCancellation: false, noiseSuppression: false, autoGainControl: false },
  });
  const Ctx = window.AudioContext || window.webkitAudioContext;
  const ctx = new Ctx();
  await ctx.resume();
  const source = ctx.createMediaStreamSource(stream);
  const analyser = ctx.createAnalyser();
  analyser.fftSize = 1024;
  analyser.smoothingTimeConstant = 0.5;
  source.connect(analyser);

  const onVisible = () => { if (document.visibilityState === 'visible' && ctx.state !== 'running') ctx.resume(); };
  document.addEventListener('visibilitychange', onVisible);

  return {
    sampleRate: ctx.sampleRate,
    fftSize: analyser.fftSize,
    readTime: buf => analyser.getFloatTimeDomainData(buf),
    readFreq: buf => analyser.getByteFrequencyData(buf),
    stop() {
      document.removeEventListener('visibilitychange', onVisible);
      stream.getTracks().forEach(t => t.stop());
      ctx.close();
    },
  };
}
```

```js
// Canvas 2D candle renderer. Draws from flame state; owns spark/smoke particles.

const WAX = '#f1e4c8', WAX_TOP = '#e2d0ad', ORANGE = '#ff7a1a', YELLOW = '#ffd36b', BLUE = '#4a7cff';

export function createRenderer(canvas) {
  const ctx = canvas.getContext('2d');
  let W = 0, H = 0;
  const sparks = [];
  const smokes = [];

  function resize() {
    const dpr = Math.min(2, window.devicePixelRatio || 1);
    W = canvas.clientWidth; H = canvas.clientHeight;
    canvas.width = Math.round(W * dpr); canvas.height = Math.round(H * dpr);
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  }

  function teardrop(x, baseY, w, h, tipDx, color, alpha) {
    ctx.beginPath();
    ctx.moveTo(x - w / 2, baseY);
    ctx.bezierCurveTo(x - w / 2, baseY - h * 0.45, x + tipDx - w * 0.15, baseY - h * 0.8, x + tipDx, baseY - h);
    ctx.bezierCurveTo(x + tipDx + w * 0.15, baseY - h * 0.8, x + w / 2, baseY - h * 0.45, x + w / 2, baseY);
    ctx.closePath();
    ctx.globalAlpha = alpha; ctx.fillStyle = color; ctx.fill(); ctx.globalAlpha = 1;
  }

  function roundedTop(x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x, y + r); ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.lineTo(x + w - r, y); ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h); ctx.lineTo(x, y + h); ctx.closePath();
  }

  function particles(list, dt) {
    for (let i = list.length - 1; i >= 0; i--) {
      const p = list[i];
      p.life -= dt / p.ttl; p.x += p.vx * dt; p.y += p.vy * dt; p.vx += p.ax * dt;
      if (p.life <= 0) list.splice(i, 1);
    }
  }

  function draw(state, intensity, time, dt, { meter = false } = {}) {
    ctx.fillStyle = '#000'; ctx.fillRect(0, 0, W, H);
    const cx = W / 2;
    const candleW = Math.min(W * 0.2, 72);
    const candleTop = H * 0.62;
    const flameBase = candleTop - 8;
    const maxH = Math.min(H * 0.22, 170);
    const flameH = maxH * state.size;
    const flicker = 0.85 + 0.15 * Math.sin(time * 23) * Math.sin(time * 7.3);
    const flameW = candleW * 0.6 * flicker * (0.6 + 0.4 * state.size);
    const tipDx = state.lean * 0.6 * flameH + Math.sin(time * 17) * 2 * state.size;
    const glow = state.lit ? state.size : 0;

    if (glow > 0.02) {
      const gy = flameBase - flameH * 0.5;
      const g = ctx.createRadialGradient(cx, gy, 0, cx, gy, maxH * 2.2);
      g.addColorStop(0, `rgba(255,150,50,${0.35 * glow})`);
      g.addColorStop(1, 'rgba(255,120,30,0)');
      ctx.fillStyle = g; ctx.fillRect(0, 0, W, H);
    }

    // candle
    ctx.fillStyle = WAX; roundedTop(cx - candleW / 2, candleTop, candleW, H - candleTop, 6); ctx.fill();
    ctx.fillStyle = WAX_TOP; ctx.beginPath(); ctx.ellipse(cx, candleTop, candleW / 2, candleW * 0.16, 0, 0, Math.PI * 2); ctx.fill();
    ctx.strokeStyle = '#1a1a1a'; ctx.lineWidth = 3; ctx.lineCap = 'round';
    ctx.beginPath(); ctx.moveTo(cx, candleTop); ctx.lineTo(cx + state.lean * 3, candleTop - 12); ctx.stroke();

    // flame
    if (state.lit && state.size > 0.01) {
      ctx.globalCompositeOperation = 'lighter';
      teardrop(cx, flameBase, flameW * 1.1, flameH, tipDx, ORANGE, 0.85);
      teardrop(cx, flameBase, flameW * 0.7, flameH * 0.72, tipDx * 0.8, YELLOW, 0.95);
      teardrop(cx, flameBase + 2, flameW * 0.5, flameH * 0.25, tipDx * 0.3, BLUE, 0.7);
      ctx.globalCompositeOperation = 'source-over';
    }

    // sparks
    if (state.lit && Math.random() < 0.08 * state.size) {
      sparks.push({ x: cx + tipDx * 0.5, y: flameBase - flameH * 0.6, vx: (Math.random() - 0.5) * 20 + state.lean * 30, vy: -40 - Math.random() * 40, ax: 0, life: 1, ttl: 0.8 + Math.random() * 0.6 });
    }
    particles(sparks, dt);
    for (const p of sparks) { ctx.globalAlpha = p.life; ctx.fillStyle = YELLOW; ctx.beginPath(); ctx.arc(p.x, p.y, 1.5, 0, Math.PI * 2); ctx.fill(); }
    ctx.globalAlpha = 1;

    // smoke
    if (state.smoke > 0 && Math.random() < 0.5) {
      smokes.push({ x: cx, y: candleTop - 14, vx: (Math.random() - 0.5) * 10, vy: -25 - Math.random() * 20, ax: (Math.random() - 0.5) * 20, life: 1, ttl: 2 + Math.random(), r: 4 + Math.random() * 6 });
    }
    particles(smokes, dt);
    for (const p of smokes) {
      ctx.globalAlpha = 0.18 * p.life * Math.max(state.smoke, 0.3);
      ctx.fillStyle = '#9a9a9a'; ctx.beginPath(); ctx.arc(p.x, p.y, p.r * (2 - p.life), 0, Math.PI * 2); ctx.fill();
    }
    ctx.globalAlpha = 1;

    if (meter) {
      const mw = Math.min(W - 32, 240);
      ctx.fillStyle = 'rgba(255,138,64,0.25)'; ctx.fillRect(16, 16, mw, 6);
      ctx.fillStyle = intensity > 0.8 ? '#ff4d1a' : '#ff8a40'; ctx.fillRect(16, 16, mw * intensity, 6);
    }
  }

  return { resize, draw };
}
```

```python
def flame_icon(size):
    rows = []
    cx = size / 2
    base_y = size * 0.80; tip_y = size * 0.12; rx_max = size * 0.26
    for y in range(size):
        row = []
        for x in range(size):
            if tip_y <= y <= base_y:
                t = (y - tip_y) / (base_y - tip_y)          # 0 at tip, 1 at base
                rx = rx_max * math.sin(math.pi * min(1.0, t * 0.5 + 0.5)) if t < 1 else 0
                rx = rx_max * (t ** 0.6) * (1 - 0.35 * t)
                d = abs(x - cx) / max(rx, 1e-6)
            else:
                d = 2
            if d > 1:
                row += [0, 0, 0]
            else:
                k = (1 - d) ** 0.8
                row += [255, int(110 + 100 * k), int(20 + 60 * k)]
        rows.append(row)
    return png(size, size, rows)
```
